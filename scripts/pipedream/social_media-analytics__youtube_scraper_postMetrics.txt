import requests
import os
from datetime import datetime, UTC

# ==============================
# 🔧 CONFIG
# ==============================

SUPABASE_URL = os.environ.get("SUPABASE_URL")
SUPABASE_KEY = os.environ.get("SUPABASE_KEY")
YOUTUBE_API_KEY = os.environ.get("YOUTUBE_API_KEY")

HEADERS = {
    "apikey": SUPABASE_KEY,
    "Authorization": f"Bearer {SUPABASE_KEY}",
    "Content-Type": "application/json"
}

TIMEOUT = 10

# ==============================
# 🔍 STEP 1 — BUSCAR FILA
# ==============================

def fetch_queue():
    print("🔍 Buscando itens na fila...")

    url = f"{SUPABASE_URL}/rest/v1/post_update_queue"

    params = {
        "select": "post_id,priority_score,next_check",
        "needs_update": "eq.true",
        "next_check": "lte.now()",
        "order": "priority_score.desc",
        "limit": 20
    }

    response = requests.get(url, headers=HEADERS, params=params, timeout=TIMEOUT)

    print("📡 Status queue:", response.status_code)

    data = response.json()

    print(f"📦 Itens encontrados: {len(data)}")

    return data


# ==============================
# 🎯 STEP 2 — EXTRAIR IDS
# ==============================

def extract_ids(rows):
    ids = [r["post_id"] for r in rows]
    print("IDs:", ids)
    return ids


# ==============================
# 🎥 STEP 3 — YOUTUBE API
# ==============================

def fetch_youtube_stats(video_ids):
    print("🌐 Chamando YouTube API...")

    url = "https://www.googleapis.com/youtube/v3/videos"

    params = {
        "part": "statistics",
        "id": ",".join(video_ids),
        "key": YOUTUBE_API_KEY
    }

    response = requests.get(url, params=params, timeout=TIMEOUT)

    print("📡 YouTube status:", response.status_code)

    data = response.json()

    return data.get("items", [])


# ==============================
# ⚙️ STEP 4 — NORMALIZAR
# ==============================

def normalize(items):
    results = []

    for video in items:
        stats = video.get("statistics", {})

        results.append({
            "post_id": video["id"],
            "views": int(stats.get("viewCount", 0)),
            "likes": int(stats.get("likeCount", 0)),
            "comments": int(stats.get("commentCount", 0)),
            "last_checked": datetime.now(UTC).isoformat()
        })

    return results


# ==============================
# 🗄️ STEP 5 — INSERT HISTORY (BATCH)
# ==============================

def insert_history_batch(records):
    print("📝 Inserindo histórico em batch")

    url = f"{SUPABASE_URL}/rest/v1/post_metrics_history"

    payload = [
        {
            "post_id": r["post_id"],
            "views": r["views"],
            "likes": r["likes"],
            "comments": r["comments"]
        }
        for r in records
    ]

    response = requests.post(url, headers=HEADERS, json=payload, timeout=TIMEOUT)

    print("📡 History batch:", response.status_code)


# ==============================
# 🔄 STEP 6 — UPDATE POSTS (BATCH)
# ==============================

def update_posts_batch(records):
    print("🔄 Atualizando posts (PATCH batch)")

    for r in records:
        url = f"{SUPABASE_URL}/rest/v1/posts?post_id=eq.{r['post_id']}"

        payload = {
            "views": r["views"],
            "likes": r["likes"],
            "comments": r["comments"]
        }

        response = requests.patch(
            url,
            headers=HEADERS,
            json=payload,
            timeout=TIMEOUT
        )

        print(f"📡 {r['post_id']}:", response.status_code)


# ==============================
# 🔁 STEP 7 — UPDATE QUEUE (BATCH)
# ==============================

def update_queue_batch(records):
    print("📌 Atualizando queue em batch")

    url = f"{SUPABASE_URL}/rest/v1/post_update_queue"

    headers = HEADERS.copy()
    headers["Prefer"] = "resolution=merge-duplicates"

    payload = [
        {
            "post_id": r["post_id"],
            "last_checked": r["last_checked"],
            "needs_update": False
        }
        for r in records
    ]

    response = requests.post(url, headers=headers, json=payload, timeout=TIMEOUT)

    print("📡 Queue batch:", response.status_code)


# ==============================
# 🚀 ENTRYPOINT
# ==============================

def run():
    print("🚀 Workflow B iniciado (Queue-based)")

    try:
        # 1. Buscar fila
        queue_items = fetch_queue()

        if not queue_items:
            print("⚠️ Nenhum item na fila.")
            return {"status": "no_data"}

        # 2. IDs
        video_ids = extract_ids(queue_items)

        # 3. YouTube API
        youtube_items = fetch_youtube_stats(video_ids)

        if not youtube_items:
            print("⚠️ Nenhum dado retornado do YouTube.")
            return {"status": "no_youtube_data"}

        # 🔥 Early exit (dados incompletos)
        if len(youtube_items) < len(video_ids):
            print("⚠️ Nem todos os vídeos retornaram do YouTube")

        # 4. Normalizar
        normalized = normalize(youtube_items)

        print(f"📊 Processando {len(normalized)} registros")

        # 5. Persistência em batch
        insert_history_batch(normalized)
        update_posts_batch(normalized)
        update_queue_batch(normalized)

        print("✅ Workflow finalizado com sucesso.")

        return {
            "status": "success",
            "processed": len(normalized)
        }

    except Exception as e:
        print("❌ ERRO DETALHADO:", str(e))
        import traceback
        traceback.print_exc()


# ==============================
# ▶️ EXECUÇÃO
# ==============================

run()
